CREATE TABLE `barangay_locks` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `barangay` varchar(100) NOT NULL,

  `admin_name` varchar(100) NOT NULL,
  `lock_time` datetime NOT NULL,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `barangay` (`barangay`),
  UNIQUE KEY `admin_id` (`admin_id`),
  KEY `lock_time` (`lock_time`),
  CONSTRAINT `barangay_locks_ibfk_1` FOREIGN KEY (`admin_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
);

🔧 API HANDLER TEMPLATE
/api/[form_name]_save.php PATTERN:
php
<?php
require_once '../includes/config.php';
require_once '../includes/functions.php';
require_once '../includes/auth.php';

require_login();

header('Content-Type: application/json');

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // CSRF Validation
    if (!verify_token($_POST['csrf_token'])) {
        echo json_encode(['success' => false, 'message' => 'Security token failed']);
        exit;
    }
    
    // Input Sanitization
    $data = [
        'field1' => sanitize($_POST['field1']),
        'field2' => intval($_POST['field2'] ?? 0)
    ];
    
    // Database Insert
    try {
        $stmt = db_query("
            INSERT INTO table_name (field1, field2, created_by) 
            VALUES (?, ?, ?)
        ", [$data['field1'], $data['field2'], $_SESSION['user_id']]);
        
        if ($stmt) {
            echo json_encode(['success' => true, 'message' => 'Data saved']);
        } else {
            echo json_encode(['success' => false, 'message' => 'Save failed']);
        }
    } catch (Exception $e) {
        echo json_encode(['success' => false, 'message' => 'Database error']);
    }
}
?>
